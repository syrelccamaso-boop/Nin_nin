CREATE TABLE public.Users (
  Email text NOT NULL,
  LastName text NOT NULL,
  FirstName text NOT NULL,
  CreatedAt timestamp with time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UserId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  Department text NOT NULL,
  ContactNumber text,
  UpdatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  RoleId bigint NOT NULL,
  AuthId uuid UNIQUE,
  Status text NOT NULL DEFAULT 'active'::text,
  CONSTRAINT Users_pkey PRIMARY KEY (UserId),
  CONSTRAINT Users_RoleId_fkey FOREIGN KEY (RoleId) REFERENCES public.Role(RoleId),
  CONSTRAINT Users_AuthId_fkey FOREIGN KEY (AuthId) REFERENCES auth.users(id)
);
CREATE TABLE public.MaintenanceHistory (
  Id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  ComputerId bigint NOT NULL,
  TechnicianId bigint,
  Description text,
  DatePerformed timestamp with time zone DEFAULT now(),
  CreatedAt timestamp with time zone,
  CONSTRAINT MaintenanceHistory_pkey PRIMARY KEY (Id),
  CONSTRAINT MaintenanceHistory_ComputerId_fkey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId),
  CONSTRAINT MaintenanceHistory_TechnicianId_fkey FOREIGN KEY (TechnicianId) REFERENCES public.Users(UserId)
);
CREATE TABLE public.ComputerHardware (
  ComputerId bigint NOT NULL,
  Processor character varying,
  Ram character varying,
  Storage character varying,
  Peripherals text,
  CreatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  HardwareId bigint NOT NULL,
  CONSTRAINT ComputerHardware_pkey PRIMARY KEY (HardwareId),
  CONSTRAINT Hardware_UnitId_fkey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId)
);
CREATE TABLE public.ComputerSoftware (
  SoftwareId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  ComputerId bigint NOT NULL,
  SoftwareName character varying NOT NULL,
  Version character varying,
  DateInstalled date,
  CreatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  UpdatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT ComputerSoftware_pkey PRIMARY KEY (SoftwareId),
  CONSTRAINT ComputerSoftware_ComputerId_ForeignKey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId)
);
CREATE TABLE public.HardwareReplacements (
  ReplacementId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  ComputerId bigint NOT NULL,
  ComponentReplaced character varying NOT NULL,
  OldComponent character varying NOT NULL,
  NewComponent character varying NOT NULL,
  DateReplaced date NOT NULL,
  Reason text NOT NULL,
  ReplacedBy bigint NOT NULL,
  CONSTRAINT HardwareReplacements_pkey PRIMARY KEY (ReplacementId),
  CONSTRAINT ReplacedParts_ComputerId_fkey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId),
  CONSTRAINT ReplacedParts_ReplacedBy_fkey FOREIGN KEY (ReplacedBy) REFERENCES public.Users(UserId)
);
CREATE TABLE public.Computers (
  ComputerId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  Room text NOT NULL,
  UnitNumber integer NOT NULL CHECK ("UnitNumber" > 0),
  Status text NOT NULL DEFAULT 'good'::text,
  EscalationStatus text NOT NULL DEFAULT 'Pending review'::text CHECK ("EscalationStatus" = ANY (ARRAY['Pending review'::text, 'Under inspection'::text, 'Cleared'::text])),
  CreatedAt timestamp with time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UpdatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT Computers_pkey PRIMARY KEY (ComputerId)
);
CREATE TABLE public.Reports (
  ReportId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  ComputerId bigint NOT NULL,
  ReportedBy bigint NOT NULL,
  ReporterName text NOT NULL DEFAULT ''::text,
  Issues ARRAY NOT NULL DEFAULT '{}'::text[],
  Comment text NOT NULL DEFAULT ''::text,
  ResolutionNote text NOT NULL DEFAULT ''::text,
  HandledBy bigint NOT NULL,
  CreatedAt timestamp with time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UpdatedAt timestamp with time zone,
  IssueId integer,
  StatusId bigint NOT NULL,
  CONSTRAINT Reports_pkey PRIMARY KEY (ReportId),
  CONSTRAINT Reports_ReporterId_ForeignKey FOREIGN KEY (ReportedBy) REFERENCES public.Users(UserId),
  CONSTRAINT Reports_ResolvedBy_ForeignKey FOREIGN KEY (HandledBy) REFERENCES public.Users(UserId),
  CONSTRAINT Reports_ComputerId_ForeignKey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId),
  CONSTRAINT Reports_IssueId_fkey FOREIGN KEY (IssueId) REFERENCES public.ReportIssues(IssueId),
  CONSTRAINT Reports_StatusId_fkey FOREIGN KEY (StatusId) REFERENCES public.ReportStatus(StatusId)
);
CREATE TABLE public.NotificationPreferences (

  UpdatedAt timestamp with time zone,

  CreatedAt timestamp with time zone DEFAULT now(),

  CONSTRAINT NotificationPreferences_UserId_fkey FOREIGN KEY (UserId) REFERENCES public.Users(UserId)

CREATE TABLE public.Notifications (
  NotificationId bigint GENERATED ALWAYS AS IDENTITY NOT NULL,

CREATE TABLE public.ReportStatus (

CREATE TABLE public.SoftwareUpdates (
  Id integer NOT NULL,
  ComputerId bigint NOT NULL,
  SoftwareName character varying NOT NULL,
  OldVersion character varying,
  NewVersion character varying NOT NULL,
  UpdatedBy bigint NOT NULL,
  UpdateDate date NOT NULL,
  CreatedAt timestamp with time zone DEFAULT CURRENT_TIMESTAMP,

  CONSTRAINT SoftwareUpdates_ComputerId_fkey FOREIGN KEY (ComputerId) REFERENCES public.Computers(ComputerId),
